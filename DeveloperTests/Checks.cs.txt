    static void BE(Stream stream, uint value) { for (int i = 3; i >= 0; i--) stream.WriteByte((byte)(value >> (i * 8))); }
    static void Chunk(Stream stream, string name, byte[] data)
    {
        byte[] type = System.Text.Encoding.ASCII.GetBytes(name);
        BE(stream, (uint)data.Length); stream.Write(type, 0, 4); stream.Write(data, 0, data.Length);
        uint crc = 0xffffffff;
        foreach (byte b in type.Concat(data)) { crc ^= b; for (int i = 0; i < 8; i++) crc = (crc >> 1) ^ ((crc & 1) != 0 ? 0xedb88320u : 0); }
        BE(stream, ~crc);
    }
    static int Paeth(int a, int b, int c) { int p=a+b-c,pa=Math.Abs(p-a),pb=Math.Abs(p-b),pc=Math.Abs(p-c); return pa<=pb&&pa<=pc?a:pb<=pc?b:c; }
    static byte[] MakePng(byte[] pixels, int width, int height, int type, int filter = 0, byte[] colours = null,
        byte[] alpha = null, int depth = 8, int interlace = 0, bool badAdler = false, bool extraInflated = false, bool split = false)
    {
        int bpp = type == 3 ? 1 : type == 2 ? 3 : 4, stride=width*bpp;
        var raw = new MemoryStream();
        for (int y=0;y<height;y++) {
            raw.WriteByte((byte)filter);
            for (int x=0;x<stride;x++) {
                int p=y*stride+x,a=x>=bpp?pixels[p-bpp]:0,b=y>0?pixels[p-stride]:0,c=x>=bpp&&y>0?pixels[p-stride-bpp]:0;
                int predictor=filter==1?a:filter==2?b:filter==3?(a+b)/2:filter==4?Paeth(a,b,c):0;
                raw.WriteByte(unchecked((byte)(pixels[p]-predictor)));
            }
        }
        if (extraInflated) raw.WriteByte(0);
        byte[] input=raw.ToArray(), z;
        using(var compressed=new MemoryStream()) {
            compressed.WriteByte(0x78);compressed.WriteByte(0x9c);
            using(var deflate=new DeflateStream(compressed,CompressionMode.Compress,true)) deflate.Write(input,0,input.Length);
            uint a=1,b=0;foreach(byte v in input){a=(a+v)%65521;b=(b+a)%65521;}
            BE(compressed,(b<<16|a)^(badAdler?1u:0u));z=compressed.ToArray();
        }
        using(var png=new MemoryStream()) {
            png.Write(new byte[]{137,80,78,71,13,10,26,10},0,8);
            using(var header=new MemoryStream()) { BE(header,(uint)width);BE(header,(uint)height);header.Write(new byte[]{(byte)depth,(byte)type,0,0,(byte)interlace},0,5);Chunk(png,"IHDR",header.ToArray()); }
            if(colours!=null)Chunk(png,"PLTE",colours);
            if(alpha!=null)Chunk(png,"tRNS",alpha);
            if(split){Chunk(png,"IDAT",z.Take(z.Length/2).ToArray());Chunk(png,"IDAT",z.Skip(z.Length/2).ToArray());}
            else Chunk(png,"IDAT",z);
            Chunk(png,"IEND",new byte[0]);return png.ToArray();
        }
    }
